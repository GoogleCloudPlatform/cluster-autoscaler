GKE Cluster Autoscaler has been deployed!

To verify that Cluster Autoscaler is running:
  kubectl get pods --namespace {{ .Release.Namespace }} -l "{{ include "cluster-autoscaler.selectorLabels" . }}"

To view autoscaler logs:
  kubectl logs --namespace {{ .Release.Namespace }} -l "{{ include "cluster-autoscaler.selectorLabels" . }}" -c {{ .Chart.Name }} -f

{{- if .Values.service.create }}
To access the metrics / health check endpoint:
{{- if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "cluster-autoscaler.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  curl http://$NODE_IP:$NODE_PORT/metrics
{{- else if contains "LoadBalancer" .Values.service.type }}
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "cluster-autoscaler.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  curl http://$SERVICE_IP:{{ .Values.service.port }}/metrics
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "{{ include "cluster-autoscaler.selectorLabels" . }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Forwarding port to access metrics at http://127.0.0.1:{{ .Values.server.port }}/metrics ..."
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME {{ .Values.server.port }}:$CONTAINER_PORT
{{- end }}
{{- end }}
